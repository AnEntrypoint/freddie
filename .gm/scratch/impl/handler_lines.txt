58|export async function runFlowSkill({ name, flowArgs = '', model = null, provider = null, ctx = null, log = null, findSkillFn = null } = {}) {
59|    if (log) log.info(`run_flow: ${name} args=${flowArgs}`)
60|
61|    let skill
62|    let findSkill = findSkillFn || null
63|    try {
64|        if (!findSkill) findSkill = (await import('../../src/skills/index.js')).findSkill
65|        skill = findSkill(name)
66|    } catch (e) {
67|        if (log) log.warn(`run_flow: skill lookup failed: ${e.message}`)
68|        skill = null
69|    }
70|
71|    if (!skill) {
72|        return { ok: false, error: `Flow skill not found: ${name}`, state: { moves: 0 } }
73|    }
74|
75|    if (!skill.body || !skill.body.trim()) {
76|        return { ok: false, error: `Skill "${name}" has no body content`, state: { moves: 0 } }
77|    }
78|
79|    const callLLM = resolveCallLLM({ provider: provider || undefined, model: model || undefined })
80|
81|    let dispatchTool = null
82|    let tools = null
83|    try {
84|        const { bootHost } = await import('../../src/host/index.js')
85|        const { getEnabledToolSchemas, getAvailableToolsets } = await import('../../src/toolsets.js')
86|        const h = await bootHost()
87|        if (h?.pi?.dispatchTool) {
88|            dispatchTool = (toolName, toolArgs, toolCtx) => {
89|                if (toolName === 'run_flow') return JSON.stringify({ error: 'run_flow cannot nest inside a flow node' })
90|                return h.pi.dispatchTool(toolName, toolArgs, toolCtx || ctx || {})
91|            }
92|        }
93|        const sets = Array.isArray(ctx?.enabledToolsets)
94|            ? ctx.enabledToolsets
95|            : await getAvailableToolsets()
96|        const disabled = [
97|            ...(Array.isArray(ctx?.disabledToolsets) ? ctx.disabledToolsets : []),
98|            'run_flow',
99|        ]
100|        const schemas = await getEnabledToolSchemas(sets, disabled)
101|        tools = (schemas || []).filter(s => (s.name || s.function?.name) !== 'run_flow')
102|        if (!ctx?.askUser) tools = tools.filter(s => (s.name || s.function?.name) !== 'ask_user_question')
103|    } catch {
104|        dispatchTool = null
105|        tools = null
106|    }
107|
108|    const findSkillBound = typeof findSkill === 'function' ? findSkill : null
109|    const runCtx = { ...(ctx || {}), findSkill: findSkillBound || ctx?.findSkill }
110|    const runner = new FlowRunner({
111|        skillName: name,
112|        skillContent: skill.body,
113|        callLLM,
114|        maxMoves: 1000,
115|        ctx: runCtx,
116|        dispatchTool,
117|        tools,
118|    })
119|
120|    const result = await runner.run(flowArgs)
121|
122|    if (log) {
123|        if (result.ok) {
124|            log.info(`run_flow: ${name} completed in ${result.state.moves} moves`)
125|        } else {
126|            log.warn(`run_flow: ${name} failed: ${result.error}`)
127|        }
128|    }
129|
130|    const out = {
131|        ok: result.ok,
132|        state: {
133|            moves: result.state.moves,
134|            path: result.state.path,
135|            nodeCount: result.state.path.length,
136|        },
137|    }
138|    if (result.result) out.result = result.result
139|    if (result.error) out.error = result.error
140|    if (result.state.filled && Object.keys(result.state.filled).length) out.state.filled = result.state.filled
141|    return out

144|export async function flowCommandHandler(state, args) {
145|    const name = args[0]
146|    if (!name) return 'usage: /flow <name> [args...] — run a flow skill'
147|
148|    const flowArgs = args.slice(1).join(' ')
149|
150|    const result = await runFlowSkill({
151|        name,
152|        flowArgs,

156|    if (!result.ok) {

163|            .join('\n')
164|        : ''
165|
166|    return `Flow "${name}" completed in ${result.state.moves} moves, ${result.state.nodeCount} nodes visited.${resSummary}`
